-- ============================================================================
-- CRIMEGRAPH: 13-TABLE RELATIONAL DBMS SCHEMA (PostgreSQL)
-- ============================================================================

DROP SCHEMA IF EXISTS public CASCADE;
CREATE SCHEMA public;

-- ----------------------------------------------------------------------------
-- 1. ENUMS & VALUE CONSTRAINTS
-- ----------------------------------------------------------------------------
CREATE TYPE user_role AS ENUM ('INVESTIGATOR', 'ADMINISTRATOR', 'AUDITOR');
CREATE TYPE case_status AS ENUM ('OPEN', 'UNDER_REVIEW', 'CLOSED');
CREATE TYPE doc_proc_status AS ENUM ('PENDING', 'PROCESSED', 'FAILED');
CREATE TYPE entity_type_enum AS ENUM ('PERSON', 'PHONE', 'VEHICLE', 'LOCATION', 'ORGANISATION');
CREATE TYPE match_review_status AS ENUM ('PENDING', 'ACCEPTED', 'REJECTED');
CREATE TYPE alert_severity_enum AS ENUM ('LOW', 'MEDIUM', 'HIGH', 'CRITICAL');
CREATE TYPE alert_status_enum AS ENUM ('NEW', 'REVIEWING', 'RESOLVED', 'DISMISSED');

-- ----------------------------------------------------------------------------
-- 2. USERS (Role-Based Access)
-- ----------------------------------------------------------------------------
CREATE TABLE users (
    user_id SERIAL PRIMARY KEY,
    username VARCHAR(64) NOT NULL UNIQUE,
    full_name VARCHAR(128) NOT NULL,
    role user_role NOT NULL DEFAULT 'INVESTIGATOR',
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- ----------------------------------------------------------------------------
-- 3. CASES & SOURCES
-- ----------------------------------------------------------------------------
CREATE TABLE cases (
    case_id SERIAL PRIMARY KEY,
    case_code VARCHAR(32) NOT NULL UNIQUE, -- e.g., 'CG-001'
    title VARCHAR(255) NOT NULL,
    description TEXT,
    status case_status NOT NULL DEFAULT 'OPEN',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE sources (
    source_id SERIAL PRIMARY KEY,
    source_name VARCHAR(128) NOT NULL,
    source_type VARCHAR(64) NOT NULL, -- 'FIR', 'CALL_LOG', 'TRANSACTION_REPORT', 'INTELLIGENCE'
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- ----------------------------------------------------------------------------
-- 4. DOCUMENTS (Unstructured Text & Extraction Staging)
-- ----------------------------------------------------------------------------
CREATE TABLE documents (
    document_id SERIAL PRIMARY KEY,
    source_id INT NOT NULL REFERENCES sources(source_id) ON DELETE RESTRICT,
    case_id INT REFERENCES cases(case_id) ON DELETE SET NULL,
    document_title VARCHAR(255) NOT NULL,
    content TEXT NOT NULL,
    processing_status doc_proc_status NOT NULL DEFAULT 'PENDING',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- ----------------------------------------------------------------------------
-- 5. ENTITIES & CASE JUNCTION
-- ----------------------------------------------------------------------------
CREATE TABLE entities (
    entity_id SERIAL PRIMARY KEY,
    entity_type entity_type_enum NOT NULL,
    display_name VARCHAR(255) NOT NULL,
    normalized_identifier VARCHAR(128), -- e.g. E.164 phone, standardized plate number
    attributes JSONB DEFAULT '{}'::jsonb, -- flexible attributes like { "alias": "Vicky", "make": "Honda" }
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE case_entities (
    case_id INT NOT NULL REFERENCES cases(case_id) ON DELETE CASCADE,
    entity_id INT NOT NULL REFERENCES entities(entity_id) ON DELETE CASCADE,
    document_id INT REFERENCES documents(document_id) ON DELETE SET NULL,
    association_type VARCHAR(64) NOT NULL DEFAULT 'MENTIONED', -- 'SUSPECT', 'WITNESS', 'MENTIONED'
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (case_id, entity_id, association_type)
);

-- ----------------------------------------------------------------------------
-- 6. RELATIONSHIPS & EVIDENCE (Graph Network & Traceability)
-- ----------------------------------------------------------------------------
CREATE TABLE relationships (
    relationship_id SERIAL PRIMARY KEY,
    source_entity_id INT NOT NULL REFERENCES entities(entity_id) ON DELETE CASCADE,
    target_entity_id INT NOT NULL REFERENCES entities(entity_id) ON DELETE CASCADE,
    relationship_type VARCHAR(64) NOT NULL, -- 'CALLED', 'ASSOCIATED_WITH', 'TRANSFERRED_TO'
    confidence NUMERIC(3, 2) NOT NULL DEFAULT 1.0 CHECK (confidence BETWEEN 0.0 AND 1.0),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT chk_no_self_relationship CHECK (source_entity_id <> target_entity_id)
);

-- Many-to-Many: One relationship can be proven by multiple documents
CREATE TABLE relationship_evidence (
    relationship_id INT NOT NULL REFERENCES relationships(relationship_id) ON DELETE CASCADE,
    document_id INT NOT NULL REFERENCES documents(document_id) ON DELETE CASCADE,
    evidence_note TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (relationship_id, document_id)
);

-- ----------------------------------------------------------------------------
-- 7. ENTITY RESOLUTION (Human-in-the-Loop Verification)
-- ----------------------------------------------------------------------------
CREATE TABLE entity_match_candidates (
    candidate_id SERIAL PRIMARY KEY,
    entity_id_1 INT NOT NULL REFERENCES entities(entity_id) ON DELETE CASCADE,
    entity_id_2 INT NOT NULL REFERENCES entities(entity_id) ON DELETE CASCADE,
    match_score NUMERIC(4, 3) NOT NULL CHECK (match_score BETWEEN 0.0 AND 1.0),
    match_reasons JSONB NOT NULL DEFAULT '{}'::jsonb, -- e.g. {"shared_phone": true, "name_levenshtein": 0.85}
    review_status match_review_status NOT NULL DEFAULT 'PENDING',
    reviewed_by INT REFERENCES users(user_id) ON DELETE SET NULL,
    reviewed_at TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT chk_distinct_candidate_entities CHECK (entity_id_1 <> entity_id_2)
);

-- ----------------------------------------------------------------------------
-- 8. STRUCTURED EVENT RECORDS (Time-Series Logs)
-- ----------------------------------------------------------------------------
CREATE TABLE call_records (
    call_id BIGSERIAL PRIMARY KEY,
    source_id INT NOT NULL REFERENCES sources(source_id) ON DELETE RESTRICT,
    case_id INT REFERENCES cases(case_id) ON DELETE SET NULL,
    caller_entity_id INT NOT NULL REFERENCES entities(entity_id) ON DELETE CASCADE,
    receiver_entity_id INT NOT NULL REFERENCES entities(entity_id) ON DELETE CASCADE,
    started_at TIMESTAMP WITH TIME ZONE NOT NULL,
    duration_seconds INT NOT NULL CHECK (duration_seconds >= 0),
    CONSTRAINT chk_distinct_callers CHECK (caller_entity_id <> receiver_entity_id)
);

CREATE TABLE transactions (
    transaction_id BIGSERIAL PRIMARY KEY,
    source_id INT NOT NULL REFERENCES sources(source_id) ON DELETE RESTRICT,
    case_id INT REFERENCES cases(case_id) ON DELETE SET NULL,
    sender_entity_id INT NOT NULL REFERENCES entities(entity_id) ON DELETE RESTRICT,
    receiver_entity_id INT NOT NULL REFERENCES entities(entity_id) ON DELETE RESTRICT,
    amount NUMERIC(14, 2) NOT NULL CHECK (amount > 0),
    currency VARCHAR(10) NOT NULL DEFAULT 'INR',
    transaction_at TIMESTAMP WITH TIME ZONE NOT NULL,
    CONSTRAINT chk_distinct_transactors CHECK (sender_entity_id <> receiver_entity_id)
);

-- ----------------------------------------------------------------------------
-- 9. ALERTS & AUDIT TRAIL
-- ----------------------------------------------------------------------------
CREATE TABLE alerts (
    alert_id SERIAL PRIMARY KEY,
    case_id INT REFERENCES cases(case_id) ON DELETE CASCADE,
    entity_id INT REFERENCES entities(entity_id) ON DELETE CASCADE,
    alert_type VARCHAR(64) NOT NULL, -- 'HIGH_CALL_FREQUENCY', 'CROSS_CASE_BRIDGE'
    description TEXT NOT NULL,
    severity alert_severity_enum NOT NULL DEFAULT 'MEDIUM',
    status alert_status_enum NOT NULL DEFAULT 'NEW',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE audit_logs (
    log_id BIGSERIAL PRIMARY KEY,
    user_id INT REFERENCES users(user_id) ON DELETE SET NULL,
    action VARCHAR(64) NOT NULL, -- 'CASE_CREATED', 'ALERT_REVIEWED', 'MATCH_CONFIRMED'
    entity_type VARCHAR(64) NOT NULL,
    record_id INT,
    details JSONB DEFAULT '{}'::jsonb,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- ----------------------------------------------------------------------------
-- 10. INDEXES (Mandatory for Rapid Traversal & Evaluator Checks)
-- ----------------------------------------------------------------------------
CREATE INDEX idx_entities_normalized ON entities(normalized_identifier);
CREATE INDEX idx_entities_type ON entities(entity_type);
CREATE INDEX idx_rel_source ON relationships(source_entity_id);
CREATE INDEX idx_rel_target ON relationships(target_entity_id);
CREATE INDEX idx_rel_evidence_doc ON relationship_evidence(document_id);
CREATE INDEX idx_case_entities_case ON case_entities(case_id);
CREATE INDEX idx_case_entities_entity ON case_entities(entity_id);
CREATE INDEX idx_calls_caller ON call_records(caller_entity_id);
CREATE INDEX idx_calls_receiver ON call_records(receiver_entity_id);
CREATE INDEX idx_calls_started_at ON call_records(started_at);
CREATE INDEX idx_transactions_sender ON transactions(sender_entity_id);
CREATE INDEX idx_transactions_receiver ON transactions(receiver_entity_id);
CREATE INDEX idx_alerts_case ON alerts(case_id);
CREATE INDEX idx_audit_user ON audit_logs(user_id);

-- ----------------------------------------------------------------------------
-- 11. AUTOMATIC AUDIT TRIGGER
-- Demonstrates stored procedure / trigger capabilities on Alert reviews
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION trg_log_alert_status_update()
RETURNS TRIGGER AS $$
BEGIN
    IF (OLD.status IS DISTINCT FROM NEW.status) THEN
        INSERT INTO audit_logs (
            user_id,
            action,
            entity_type,
            record_id,
            details
        ) VALUES (
            NULL, -- System/Trigger context (or pass user context via session)
            'ALERT_STATUS_UPDATED',
            'alerts',
            NEW.alert_id,
            json_build_object(
                'previous_status', OLD.status,
                'new_status', NEW.status,
                'alert_type', NEW.alert_type
            )::jsonb
        );
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_alert_status_audit
AFTER UPDATE OF status ON alerts
FOR EACH ROW
EXECUTE FUNCTION trg_log_alert_status_update();
